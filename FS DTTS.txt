                              <fs_item>-batch <fs_item>-exp_date

    lwa_product_line-xd   = lv_exp_date_formatted.

    APPEND lwa_product_line TO lwa_request-deactivation_request-productlist-product.

  ENDLOOP.
  SHIFT lv_grund LEFT DELETING LEADING '0'.

  LOOP AT p_response-accept_batch_service_response-productlist-product
       INTO lwa_product_resp.

      lv_item_gtin = <fs_item>-gtin.
      lv_resp_gtin = lwa_product_resp-gtin.

  AND <fs_item>-batch = lwa_product_resp-bn
  AND <fs_item>-prod_qty = lwa_product_resp-quantity.

        ENDIF.

        lv_error_code = lwa_product_resp-rc.
        PERFORM get_error_description
          USING lv_error_code
          CHANGING lv_description.

        <fs_item>-trans_stat  = lv_description.
        <fs_item>-changed_date = sy-datum.
        <fs_item>-changed_time = sy-uzeit.
        <fs_item>-changed_by   = sy-uname.
        EXIT.
      ENDIF.
    ENDLOOP.

  ENDLOOP.

  LOOP AT p_header_table ASSIGNING <fs_header>.
    IF lv_all_success = abap_true.
      <fs_header>-status = 'SUCCESS'.
    ELSE.
      <fs_header>-status = 'ERROR'.
    ENDIF.

    <fs_header>-changed_time = sy-uzeit.

*&---------------------------------------------------------------------*
*& SECTION 4: EXCEPTION HANDLERS & SUPPORTING PERFORMS
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*& Form  HANDLE_DTTS_APPLICATION_FAULT
*&---------------------------------------------------------------------*
*  Generic handler using config table fault attribute
FORM handle_dtts_application_fault USING p_exception TYPE REF TO cx_ai_application_fault
                                         p_operation TYPE char20
                                   CHANGING p_items_table TYPE table
                                            p_header_table TYPE table.

  DATA: lv_response_code TYPE string,
        lv_message       TYPE string.

  FIELD-SYMBOLS: <fs_item>   TYPE zmm_sst_dtts_itm,
                 <fs_header> TYPE zmm_sst_dtts_hdr,
                 <fs_field>  TYPE any.

  " ========== Extract Fault Details (Direct Access - Flat Structure) ==========

  TRY.
      " Try to access RESPONSE_CODE directly
      ASSIGN p_exception->('RESPONSE_CODE') TO <fs_field>.
      IF sy-subrc = 0 AND <fs_field> IS ASSIGNED.
        lv_response_code = <fs_field>.
      ENDIF.

      " Try to access MESSAGE directly
      UNASSIGN <fs_field>.
      ASSIGN p_exception->('MESSAGE') TO <fs_field>.
      IF sy-subrc = 0 AND <fs_field> IS ASSIGNED.
        lv_message = <fs_field>.
      ENDIF.

    CATCH cx_root.
      " Continue to fallback
  ENDTRY.

  " ========== Fallback if extraction failed ==========

  IF lv_response_code IS INITIAL.
    lv_response_code = 'APP_FAULT'.
  ENDIF.

  IF lv_message IS INITIAL.
    lv_message = p_exception->get_text( ).
    IF lv_message IS INITIAL.
      lv_message = |DTTS Application Fault - Operation: { p_operation }|.
    ENDIF.
  ENDIF.

  " ========== Update ALL Items with Fault Details ==========

  LOOP AT p_items_table ASSIGNING <fs_item>.
    <fs_item>-tr_response = lv_response_code.
    <fs_item>-trans_stat  = lv_message.
    <fs_item>-prod_stat   = 'ERROR'.
    <fs_item>-notif_id    = ''.

    " Update item timestamp
    <fs_item>-changed_date = sy-datum.
    <fs_item>-changed_time = sy-uzeit.
    <fs_item>-changed_by   = sy-uname.
  ENDLOOP.

  " Update headers with timestamp and STATUS
  LOOP AT p_header_table ASSIGNING <fs_header>.
    <fs_header>-status = 'ERROR'.
    <fs_header>-changed_date = sy-datum.
    <fs_header>-changed_time = sy-uzeit.
    <fs_header>-changed_by   = sy-uname.
  ENDLOOP.

  " Save to database
  IF p_items_table IS NOT INITIAL.
    MODIFY zmm_sst_dtts_itm FROM TABLE p_items_table.
  ENDIF.

  IF p_header_table IS NOT INITIAL.
    MODIFY zmm_sst_dtts_hdr FROM TABLE p_header_table.
  ENDIF.

ENDFORM.



*&---------------------------------------------------------------------*
*& Form  HANDLE_API_ERROR_MULTIPLE
*&---------------------------------------------------------------------*
FORM handle_api_error_multiple USING p_exception TYPE REF TO cx_ai_system_fault
                               CHANGING p_items_table TYPE table
                                           p_header_table TYPE table.

  DATA: lv_sys_error  TYPE string,
        lv_mpl_id     TYPE string,
        lv_error_code TYPE string.

  FIELD-SYMBOLS: <fs_item>   TYPE zmm_sst_dtts_itm,
                 <fs_header> TYPE zmm_sst_dtts_hdr.

  " Get error message
  lv_sys_error = p_exception->get_text( ).

  " Extract MPL ID if available
  IF lv_sys_error CS 'MPL ID'.
    FIND REGEX 'MPL ID ([A-Za-z0-9_]+)' IN lv_sys_error
         SUBMATCHES lv_mpl_id.
  ENDIF.

  " Extract DTTS error code if available (e.g., 98765)
  IF lv_sys_error CS '<FC>'.
    FIND REGEX '<FC>(\d+)</FC>' IN lv_sys_error
         SUBMATCHES lv_error_code.
  ENDIF.

  " Update ALL items with error code
  LOOP AT p_items_table ASSIGNING <fs_item>.
    <fs_item>-tr_response = lv_error_code.
    <fs_item>-prod_stat = 'ERROR'.
    <fs_item>-trans_stat  = lv_sys_error.
    " Update item timestamp
    <fs_item>-changed_date = sy-datum.
    <fs_item>-changed_time = sy-uzeit.
    <fs_item>-changed_by   = sy-uname.
  ENDLOOP.

  "  Update headers
  LOOP AT p_header_table ASSIGNING <fs_header>.
    <fs_header>-status = 'ERROR'.
    <fs_header>-changed_date = sy-datum.
    <fs_header>-changed_time = sy-uzeit.
    <fs_header>-changed_by   = sy-uname.
  ENDLOOP.

  " Save  error records to database
  MODIFY zmm_sst_dtts_itm FROM TABLE p_items_table.
  MODIFY zmm_sst_dtts_hdr FROM TABLE p_header_table.

ENDFORM.

FORM get_gtin USING p_matnr TYPE matnr
                    p_meins TYPE meins
              CHANGING p_gtin TYPE z_dgtin.

  DATA: lwa_mean      TYPE mean,
        lwa_marm      TYPE marm,
        lv_base_meins TYPE meins.

  CLEAR p_gtin.

  " Try MEAN table first
  SELECT SINGLE * FROM mean WHERE matnr = @p_matnr
      AND meinh = @p_meins INTO @lwa_mean.

  IF sy-subrc = 0 AND lwa_mean-ean11 IS NOT INITIAL.
    p_gtin = lwa_mean-ean11.
  ELSE.
    " Try MARM table
    SELECT SINGLE * FROM marm WHERE matnr = @p_matnr
        AND meinh = @p_meins INTO @lwa_marm.

    IF sy-subrc = 0 AND lwa_marm-ean11 IS NOT INITIAL.
      p_gtin = lwa_marm-ean11.
    ENDIF.
  ENDIF.
  IF p_gtin IS INITIAL.
    CLEAR : lwa_mean,lwa_marm.
    SELECT SINGLE meins FROM mara WHERE matnr = @p_matnr INTO @lv_base_meins.
    IF sy-subrc = 0 AND lv_base_meins <> p_meins.

      SELECT SINGLE * FROM mean WHERE matnr = @p_matnr
AND meinh = @lv_base_meins INTO @lwa_mean.

      IF sy-subrc = 0 AND lwa_mean-ean11 IS NOT INITIAL.
        p_gtin = lwa_mean-ean11.
      ELSE.
        " Try MARM table
        SELECT SINGLE * FROM marm WHERE matnr = @p_matnr
            AND meinh = @lv_base_meins INTO @lwa_marm.

        IF sy-subrc = 0 AND lwa_marm-ean11 IS NOT INITIAL.
          p_gtin = lwa_marm-ean11.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form  FORMAT_DATA
*&---------------------------------------------------------------------*
FORM format_data USING p_gtin_in TYPE any
                       p_quantity_in TYPE any
                       p_batch_in TYPE any
                       p_exp_date_in TYPE any
                 CHANGING p_gtin_out TYPE string
                          p_quantity_out TYPE string
                          p_batch_out TYPE string
                          p_exp_date_out TYPE string.

  DATA: lv_len      TYPE i,
        lv_qty_int  TYPE string,
        lv_qty_dec  TYPE string,
        lv_year(4)  TYPE c,
        lv_month(2) TYPE c,
        lv_day(2)   TYPE c.

  " Format GTIN
  p_gtin_out = p_gtin_in.
  CONDENSE p_gtin_out NO-GAPS.
  lv_len = strlen( p_gtin_out ).
  IF lv_len < 14.
    p_gtin_out = |{ p_gtin_out WIDTH = 14 PAD = '0' ALIGN = RIGHT }|.
  ENDIF.

  " Format Quantity
  p_quantity_out = p_quantity_in.
  IF p_quantity_out CS '.'.
    SPLIT p_quantity_out AT '.' INTO lv_qty_int lv_qty_dec.
    p_quantity_out = lv_qty_int.
  ENDIF.
  p_quantity_out = |{ p_quantity_out ALPHA = OUT }|.
  CONDENSE p_quantity_out NO-GAPS.

  " Format Batch
  p_batch_out = p_batch_in.
  CONDENSE p_batch_out NO-GAPS.

  " Format Expiry Date
  CLEAR p_exp_date_out.
  IF p_exp_date_in IS NOT INITIAL.
    lv_year  = p_exp_date_in+0(4).
    lv_month = p_exp_date_in+4(2).
    lv_day   = p_exp_date_in+6(2).
    p_exp_date_out = |{ lv_year }-{ lv_month }-{ lv_day }|.
  ENDIF.

ENDFORM.

  SELECT SINGLE zzgln FROM zmm_migo_hdrcust
    WHERE mblnr = @p_mseg-mblnr
      AND mjahr = @p_mseg-mjahr
    INTO @DATA(lv_custom_gln).

    CASE p_mseg-bwart.

      WHEN '101' OR '105' OR 'Z01' OR 'Z07'   "Accept
        OR '351'                               "Dispatch/Transfer
        OR '201'                               "Consume/Drug Sale
        OR '551' OR 'Z05'                      "Deactivate
        OR 'Z03'.                              "Transfer
        "*" Forward movement: custom GLN goes to FROM GLN
        p_from_gln = lv_custom_gln.
        PERFORM get_gln_type_from_value USING p_from_gln
                                        CHANGING p_from_gln_type.

      WHEN '102' OR '106' OR 'Z02' OR 'Z08' OR '122'  "Return
        OR '352'                                        "Dispatch/Transfer Cancel
        OR '202'                                        "Consume/Drug Sale Cancel
        OR '552' OR 'Z06'                               "Deactivate Cancel
        OR 'Z04'.                                       "Transfer Cancel
        "*" Reversal movement: custom GLN goes to TO GLN
        p_to_gln = lv_custom_gln.
        PERFORM get_gln_type_from_value USING p_to_gln
                                        CHANGING p_to_gln_type.

      WHEN OTHERS.
        "*" Unknown movement type: default to FROM GLN (safe fallback)
        p_from_gln = lv_custom_gln.
        PERFORM get_gln_type_from_value USING p_from_gln
                                        CHANGING p_from_gln_type.

    ENDCASE.

  ENDIF.

  CASE p_mseg-bwart.

*------------------------------------------------------------------*

                                      'P'
                                CHANGING p_to_gln
                                         p_to_gln_type.
      ENDIF.

*------------------------------------------------------------------*
*  Z06 - Deactivate Unit Cancel
*------------------------------------------------------------------*
    WHEN 'Z06'.

      IF p_from_gln IS INITIAL.
        IF p_mseg-lifnr IS NOT INITIAL.
          PERFORM get_partner_gln USING p_mseg-lifnr
                                        'P'
                                  CHANGING p_from_gln
                                           p_from_gln_type.
        ENDIF.
      ENDIF.

      PERFORM get_warehouse_gln USING p_mseg-werks
                                      p_mseg-lgort
                                CHANGING p_to_gln
                                         p_to_gln_type.

*------------------------------------------------------------------*
*  GENERIC FALLBACK
*------------------------------------------------------------------*
    WHEN OTHERS.

      IF p_from_gln IS INITIAL.
        IF p_mseg-kunnr IS NOT INITIAL.
          PERFORM get_partner_gln USING p_mseg-kunnr
                                        'C'
                                  CHANGING p_from_gln
                                           p_from_gln_type.
        ELSEIF p_mseg-lifnr IS NOT INITIAL.
          PERFORM get_partner_gln USING p_mseg-lifnr
                                        'P'
                                  CHANGING p_from_gln
                                           p_from_gln_type.
        ELSEIF p_mseg-ebeln IS NOT INITIAL.
          SELECT SINGLE werks, lgort, reslo
            FROM ekpo
            WHERE ebeln = @p_mseg-ebeln
              AND ebelp = @p_mseg-ebelp
            INTO CORRESPONDING FIELDS OF @lwa_ekpo.

          IF sy-subrc = 0 AND lwa_ekpo-reslo IS NOT INITIAL.
            PERFORM get_warehouse_gln USING lwa_ekpo-werks
                                            lwa_ekpo-reslo
                                      CHANGING p_from_gln
                                               p_from_gln_type.
          ENDIF.
        ELSE.
          PERFORM get_warehouse_gln USING p_mseg-werks
                                          p_mseg-lgort
                                    CHANGING p_from_gln
                                             p_from_gln_type.
        ENDIF.
      ENDIF.

      IF p_mseg-ebeln IS NOT INITIAL AND lwa_ekpo IS NOT INITIAL.
        IF lwa_ekpo-lgort IS NOT INITIAL
           AND ( lwa_ekpo-reslo IS INITIAL OR lwa_ekpo-lgort <> lwa_ekpo-reslo ).
          PERFORM get_warehouse_gln USING lwa_ekpo-werks
                                          lwa_ekpo-lgort
                                    CHANGING p_to_gln
                                             p_to_gln_type.
        ENDIF.
      ENDIF.

      IF p_to_gln IS INITIAL AND p_mseg-umwrk IS NOT INITIAL.
        IF p_mseg-umwrk <> p_mseg-werks OR p_mseg-umlgo <> p_mseg-lgort.
          PERFORM get_warehouse_gln USING p_mseg-umwrk
                                          p_mseg-umlgo
                                    CHANGING p_to_gln
                                             p_to_gln_type.
        ENDIF.
      ENDIF.

  ENDCASE.


ENDFORM.

*&---------------------------------------------------------------------*
*& Form  GET_PARTNER_GLN
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
FORM get_partner_gln USING    p_partner TYPE any
                              p_type TYPE char1
                     CHANGING p_gln TYPE zmm_br_gln
                              p_gln_type TYPE char1.

  CLEAR: p_gln, p_gln_type.

  CHECK p_partner IS NOT INITIAL.

  SELECT SINGLE gln
    FROM zmm_gln_bp
    WHERE bp = @p_partner
      AND zact = 'X'
    INTO @p_gln.

  IF sy-subrc = 0 AND p_gln IS NOT INITIAL.
    p_gln_type = p_type.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form  GET_WAREHOUSE_GLN
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
FORM get_warehouse_gln USING    p_werks TYPE werks_d
                                p_lgort TYPE lgort_d
                       CHANGING p_gln TYPE zmm_br_gln
                                p_gln_type TYPE char1.

  CLEAR: p_gln, p_gln_type.

  CHECK p_werks IS NOT INITIAL AND p_lgort IS NOT INITIAL.

  SELECT SINGLE gln, gln_type
    FROM zmmgln
    WHERE werks = @p_werks
      AND lgort = @p_lgort
    INTO (@p_gln, @p_gln_type).

ENDFORM.

*&---------------------------------------------------------------------*
*& Form  VALIDATE_AND_GET_OPERATION
*&---------------------------------------------------------------------*
FORM validate_and_get_operation USING p_movement_type TYPE bwart
                                      p_from_gln_type TYPE char1
                                      p_to_gln_type TYPE char1
                                CHANGING p_valid TYPE abap_bool
                                         p_operation TYPE char20.



  " Initialize
  p_valid = abap_false.
  CLEAR p_operation.

  CASE p_movement_type.

      " ========== FORWARD MOVEMENTS ==========

      " ACCEPT - Movement types: 101, 105, Z01, Z07
    WHEN '101' OR '105' OR 'Z01' OR 'Z07'.
      IF p_to_gln_type IS NOT INITIAL.
        p_operation = 'ACCEPT'.
        p_valid = abap_true.
      ENDIF.

      " DISPATCH (Wholesaler) or TRANSFER (Clinic/Pharmacy)
      " Movement type: 351
    WHEN '351'.
      IF p_to_gln_type IS NOT INITIAL.
        " Use FROM GLN type (warehouse source) to determine operation
        IF p_from_gln_type = 'W'.
          p_operation = 'DISPATCH'.
          p_valid = abap_true.
        ELSEIF p_from_gln_type = 'C' OR p_from_gln_type = 'P'.
          p_operation = 'TRANSFER'.
          p_valid = abap_true.
        ELSE.
          " Defaulting to TRANSFER if GLN type unknown
          p_operation = 'TRANSFER'.
          p_valid = abap_true.
        ENDIF.
      ENDIF.

      " CONSUME (Clinic) or DRUG_SALE (Pharmacy)
      " Movement type: 201
    WHEN '201'.
      IF p_from_gln_type = 'C'.
        p_operation = 'CONSUME'.
        p_valid = abap_true.
      ELSEIF p_from_gln_type = 'P'.
        p_operation = 'DRUG_SALE'.
        p_valid = abap_true.
      ELSE.
        " Default to CONSUME if GLN type not determined
        p_operation = 'CONSUME'.
        p_valid = abap_true.
      ENDIF.

      " DEACTIVATE - Movement types: 551, Z05
    WHEN '551' OR 'Z05'.
      p_operation = 'DEACTIVATE'.
      p_valid = abap_true.

      " Dispatch - Movement type: Z03
    WHEN 'Z03'.
   IF p_from_gln_type = 'W'.
        p_operation = 'DISPATCH'.
        p_valid = abap_true.
      ELSEIF p_from_gln_type = 'C'.
        p_operation = 'TRANSFER'.
        p_valid = abap_true.
      ELSE.
        p_operation = 'DISPATCH'.
        p_valid = abap_true.
      ENDIF.

      " ========== REVERSAL MOVEMENTS ==========

      " RETURN - Movement types: 102, 106, Z02, Z08, 122
    WHEN '102' OR '106' OR 'Z02' OR 'Z08' OR '122'.
      IF p_to_gln_type IS NOT INITIAL.
        p_operation = 'RETURN'.
        p_valid = abap_true.
      ENDIF.

      " DISPATCH_CANCEL (Wholesaler) or TRANSFER_CANCEL (Clinic/Pharmacy)
      " Movement type: 352
      "REVERSAL MOVEMENT:  check TO_GLN_TYPE!
    WHEN '352'.
      IF p_to_gln_type IS NOT INITIAL.
        "  Use TO GLN type (warehouse destination) for reversal
        IF p_to_gln_type = 'W'.
          p_operation = 'DISPATCH_CANCEL'.
          p_valid = abap_true.
        ELSEIF p_to_gln_type = 'C' OR p_to_gln_type = 'P'.
          p_operation = 'TRANSFER_CANCEL'.
          p_valid = abap_true.
        ELSE.
          " Default to TRANSFER_CANCEL
          p_operation = 'TRANSFER_CANCEL'.
          p_valid = abap_true.
        ENDIF.
      ENDIF.

      " CONSUME_CANCEL (Clinic) or DRUG_SALE_CANCEL (Pharmacy)
      " Movement type: 202
    WHEN '202'.
      " Use TO GLN type (warehouse destination) for reversal
      IF p_to_gln_type = 'C'.
        p_operation = 'CONSUME_CANCEL'.
        p_valid = abap_true.
      ELSEIF p_to_gln_type = 'P'.
        p_operation = 'DRUG_SALE_CANCEL'.
        p_valid = abap_true.
      ELSE.
        " Defaulting to CONSUME_CANCEL if GLN type not determined
        p_operation = 'CONSUME_CANCEL'.
        p_valid = abap_true.
      ENDIF.

      " DEACTIVATE_CANCEL - Movement types: 552, Z06
    WHEN '552' OR 'Z06'.
      p_operation = 'DEACTIVATE_CANCEL'.
      p_valid = abap_true.

      " DISPATCH_CANCEL - Movement type: Z04
    WHEN 'Z04'.
       IF p_to_gln_type = 'W'.
        p_operation = 'DISPATCH_CANCEL'.
        p_valid = abap_true.
      ELSEIF p_to_gln_type = 'C'.
        p_operation = 'TRANSFER_CANCEL'.
        p_valid = abap_true.
      ELSE.
        p_operation = 'DISPATCH_CANCEL'.
        p_valid = abap_true.
      ENDIF.

      " ========== UNMAPPED MOVEMENT TYPES ==========
    WHEN OTHERS.
      p_valid = abap_false.
      CLEAR p_operation.

  ENDCASE.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form  GET_GLN_TYPE_FROM_VALUE
*&---------------------------------------------------------------------*
FORM get_gln_type_from_value USING p_gln TYPE zmm_br_gln
                             CHANGING p_gln_type TYPE char1.

  CLEAR p_gln_type.

  CHECK p_gln IS NOT INITIAL.

  " Try warehouse table first
  SELECT SINGLE gln_type FROM zmmgln
    WHERE gln = @p_gln
    INTO @p_gln_type.

  IF sy-subrc = 0.
    " Found in warehouse table
    RETURN.
  ENDIF.

  " Trying partner/vendor table
  SELECT SINGLE gln FROM zmm_gln_BP
    WHERE gln = @p_gln
    INTO @DATA(lv_dummy).

  IF sy-subrc = 0.
    p_gln_type = 'P'.
    RETURN.
  ENDIF.

  " Default to warehouse type
  p_gln_type = 'W'.

ENDFORM.
